REDE_SCHEMECODE= (SELECT BIDE_SCHEMECODE from kmchdc_user.HS_BIDE_BILLDETAIL where BIDE_BILLDETAILID=REDE_BILLDETAILID AND BIDE_RATETYPE='SSF')
