Declare 
    vTestNameID Number;
Begin
    For Rec in (select TESTNAMECODE,INSCODE,upper(TESTNAME)TESTNAME,GENTESTPRICE testprice,DEPID,TESTNAMECATAGORY,ORGMULTIBILLCATEGORY,
TENA_ACCHEADID ACCHEADID,REVENUECATEGORYID,REVENUESUBCATEGORYID
 from testnamelistnew where depid is not null )
Loop
    Select Max(TENA_TestNameID)+1 Into  vTestNameID From HS_TENA_TEstName;
    Insert Into HS_TENA_TEstName(TENA_TestNameID,TENA_DepId,TENA_TestNameCode,TENA_TestName,TENA_IsActive,TENA_TNCATEGORYCODE,
    TENA_DATAPOSTBY,TENA_DATAPOSTDATE,TENA_DATAPOSTTIME,TENA_REPORTINGTYPE,TENA_ORGMULTIBILLCATEGORY,TENA_ISBLKSERVICE,
    TENA_INSCODE,TENA_INSTESTNAME,TENA_REVENUECATEGORYID,TENA_REVENUESUBCATEGORYID,TENA_ACCHEADID ,TENA_REMARKS
        ) Values (vTestNameID,REC.DEPID,Rec.TEstNameCode,Rec.TESTNAME,'Y',rec.TESTNAMECATAGORY,1,'2079/05/02','10:10:10',
        'LE',rec.ORGMULTIBILLCATEGORY,'N',Rec.INSCODE,Rec.TESTNAME,rec.REVENUECATEGORYID,rec.REVENUESUBCATEGORYID,REC.ACCHEADID,'IMPORTED FROM EXCEL' );
     Insert into HS_TENP_TestNamePrice(TENP_TESTNAMEPRICEID,TENP_TESTNAMEID,TENP_PATIENTTYPECODE,TENP_DEPID,TENP_CURRENCYTYPE,
TENP_TESTPRICE,TENP_VATAMT,TENP_DATAPOSTBY,TENP_DATAPOSTDATE,TENP_DATAPOSTTIME) Values ((Select Max(TENP_TESTNAMEPRICEID)+1 From HS_TENP_TestNamePrice),vTestNameID,'SSF',REC.DEPID,'NRS',
REC.testprice,0,1,'2079/05/02','10:10:10');
End Loop;
End;
