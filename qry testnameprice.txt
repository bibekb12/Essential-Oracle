select row_number() over (order by tena_testname) SN 
,dept_depname department, tena_testnamecode testcode,
tena_testname testname,
(select tenp_testprice from hs_tenp_testnameprice where tenp_testnameid=tena_testnameid and tenp_patienttypecode='GEN') generalrate
,(select tenp_testprice from hs_tenp_testnameprice where tenp_testnameid=tena_testnameid and tenp_patienttypecode='EHS') ehsrate
,(select TENP_ISVATABLE from hs_tenp_testnameprice where tenp_testnameid=tena_testnameid and tenp_patienttypecode='GEN') GENISVATABLE
,(select TENP_ISVATABLE from hs_tenp_testnameprice where tenp_testnameid=tena_testnameid and tenp_patienttypecode='EHS') EHSISVATABLE
,(select TENP_ISDISCOUNTABLE from hs_tenp_testnameprice where tenp_testnameid=tena_testnameid and tenp_patienttypecode='GEN') GENISVATABLE
,(select TENP_ISDISCOUNTABLE from hs_tenp_testnameprice where tenp_testnameid=tena_testnameid and tenp_patienttypecode='EHS') EHSISVATABLE
,(select TENP_ISVATABLE from hs_tenp_testnameprice where tenp_testnameid=tena_testnameid and tenp_patienttypecode='GEN') GENISVATABLE
,(select TENP_ISVATABLE from hs_tenp_testnameprice where tenp_testnameid=tena_testnameid and tenp_patienttypecode='EHS') EHSISVATABLE
from hs_tena_testname
join hs_dept_department on dept_depid= tena_depid
where dept_depid=49;
