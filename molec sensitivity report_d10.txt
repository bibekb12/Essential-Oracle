CREATE OR REPLACE procedure BIRHOS_USER.CreateSensitivityReportView IS
    i integer;
    prev_medicine clob;
    x_query clob;
    c_sensitive_fields clob;
    c_sensitive_query clob;
    var_institution varchar2(50);
BEGIN
    i := 0;
    var_institution := 'BIR HOSPITAL';
    FOR rec in(
    SELECT  'MAX(DECODE(s3.SENSITIVEMEDICINEID ,'''|| SENSITIVEMEDICINEID ||''', s."RESULT")) AS "'||MEDICINE||'"'   aa ,
    '"'||MEDICINE||'"' MEDICINE_FIELDS, MEDICINE
    FROM SENSITIVEMEDICINE s
    WHERE SENSITIVEMEDICINEID IN (SELECT SENSITIVEMEDICINEID FROM SENSITIVEMEDICINETESTWISE)
    AND MEDICINE IS NOT NULL
    ORDER BY MEDICINE)
    LOOP 
        IF rec.MEDICINE = prev_medicine THEN
            i := i+1;
            c_sensitive_query := c_sensitive_query||','||REPLACE(rec.aa,rec.medicine,rec.medicine||i);    
            c_sensitive_fields := c_sensitive_fields||','||REPLACE(rec.MEDICINE_FIELDS,rec.medicine,rec.medicine||i); 
        ELSE
            i := 0;
            c_sensitive_query := c_sensitive_query||','||rec.aa;    
            c_sensitive_fields := c_sensitive_fields||','||rec.MEDICINE_FIELDS;
        END IF;
        prev_medicine := rec.medicine;
    END LOOP;
    x_query := 'SELECT PAMA_PATIENTID PATIENTID,PAMA_PATIENTNAME PATIENTNAME, address,AGE,PAMA_GENDER GENDER,INSTITUTION,IPDOPD,DEPARTMENT
            ,REGADMDATE,SPECIMINE,COLLECTEDDATE,FINDINGS'||c_sensitive_fields||'
            FROM 
            (SELECT hpp.PAMA_PATIENTID , hpp.PAMA_PATIENTNAME , (select districtname from hs_district where districtid=pama_districtid) address 
            , PAMA_AGE||'' ''||substr(PAMA_AGETYPE,1,1) AGE , hpp.PAMA_GENDER , '''||var_institution||''' AS institution,
            CASE WHEN p2.INPATIENTID >0 THEN ''IPD''
                                   ELSE ''OPD'' END AS IPDOPD,
             (SELECT dept_depname FROM HS_DEPT_DEPARTMENT hdd WHERE hdd.DEPT_DEPID=p2.REFDEPID)
                                   DEPARTMENT      ,
             COALESCE((SELECT hii.INRE_ADMNDATE FROM HS_INRE_INPATIENTREG hii WHERE hii.INRE_INPATIENTID=p2.INPATIENTID)
                                   ,hpp.PAMA_REGDATE) REGADMDATE,
             (SELECT htt2.TENA_TESTNAME FROM HS_TENA_TESTNAME htt2  WHERE htt2.TENA_TESTNAMEID = p.TESTNAMEID) SPECIMINE,
             (SELECT s.SAMPLETAKENDATE||'' ''||s.SAMPLETAKENTIME FROM SAMPLECOLLECTION s 
             WHERE s.PATIENTTESTID = p.PATIENTTESTID) COLLECTEDDATE,
                                   tab_to_string(CAST(COLLECT(DISTINCT FINDING) AS t_varchar2_tab)) findings,
                                   p.PATIENTTESTID , p.REPNO 
            FROM 
            PATIENTTESTDETAIL p 
            JOIN PATIENTTEST p2 ON p2.PATIENTTESTID = p.PATIENTTESTID 
            JOIN HS_PAMA_PATIENTMAIN hpp ON hpp.PAMA_PATIENTID = p.PATIENTID 
            WHERE p2.TESTNAMEID IN (
            SELECT tena_testnameid FROM HS_TENA_TESTNAME htt  WHERE htt.TENA_TESTNAME LIKE ''%C/S%''
            )
            AND p.FINDING IS NOT NULL 
            GROUP BY  hpp.PAMA_PATIENTID, hpp.PAMA_PATIENTNAME , hpp.PAMA_AGE ,hpp.pama_districtid,hpp.pama_agetype,
            hpp.PAMA_GENDER,p2.INPATIENTID,p2.REFDEPID,hpp.PAMA_REGDATE,p.PATIENTTESTID,p.TESTNAMEID,p.REPNO ) R
            JOIN (select ptd.PATIENTTESTID qPATIENTTESTID, ptd.REPNO  qREPNO'||c_sensitive_query
            ||' From patienttestdetail ptd
            JOIN SENSITIVITYRESULT s ON s.patienttestid = ptd.PATIENTTESTID AND s.REPNO = ptd.REPNO 
            JOIN SENSITIVEMEDICINETESTWISE s2 ON s2.SMTWID = s.SMTWID 
            JOIN SENSITIVEMEDICINE s3 ON s3.SENSITIVEMEDICINEID = s2.SENSITIVEMEDICINEID 
            WHERE FINDING IS NOT NULL 
            GROUP BY ptd.PATIENTTESTID,ptd.REPNO) Q
            ON q.qPATIENTTESTID = r.PATIENTTESTID 
            AND q.qrepno = r.repno';
--           DBMS_OUTPUT.PUT_LINE(x_query);
             EXECUTE IMMEDIATE 'Create or replace view vw_lb_sensitivity_report as '||x_query;
    EXCEPTION
    WHEN OTHERS THEN
   raise_application_error(-20001,'An error was encountered - '||SQLCODE||' -ERROR- '||SQLERRM);    
END;
/
