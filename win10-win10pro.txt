# Check if running as administrator on Powershell
$isAdmin = ([Security.Principal.WindowsPrincipal] [Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
if (-not $isAdmin) {
    Write-Host "Please run this script as an administrator."
    exit
}

# Define the Windows 10 Pro default key
$proKey = "VK7JG-NPHTM-C97JM-9MPGT-3V66T"

# Attempt to upgrade to Windows 10 Pro
Write-Host "Starting upgrade to Windows 10 Pro..."
Invoke-Expression "cmd /c start /wait c:\windows\system32\changepk.exe /ProductKey $proKey"

# Check if the upgrade was successful
$proInstalled = (Get-WmiObject -Class SoftwareLicensingProduct | Where-Object { $_.ID -like "*Windows(R) Operating System, VOLUME_*" }).LicenseStatus
if ($proInstalled -eq 1) {
    Write-Host "Windows 10 Pro upgrade was successful!"
} else {
    Write-Host "Windows 10 Pro upgrade was not successful."
}

Read-Host "Press Enter to exit..."
